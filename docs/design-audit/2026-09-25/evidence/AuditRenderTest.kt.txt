/** One ADR-032 window, in dp. The `@Config(qualifiers)` on each test method must match it. */
internal data class AuditSize(val w: Int, val h: Int)

internal val AUDIT_360 = AuditSize(360, 640)
internal val AUDIT_412 = AuditSize(412, 915)
internal val AUDIT_800 = AuditSize(800, 360)
internal val AUDIT_600 = AuditSize(600, 960)

internal val AUDIT_FONT_SCALES = listOf(1f, 1.6f, 2f)
internal const val AUDIT_WAIT_MS = 8_000L
internal const val AUDIT_POLL_MS = 50L
internal const val AUDIT_OUT_ROOT = "build/screen-renders/audit-2026-09-25"
internal const val AUDIT_NEVER_READY = "never-ready.txt"
internal const val AUDIT_MANIFEST = "manifest.csv"
internal const val AUDIT_MANIFEST_HEADER = "file,screen,state,width,height,font,bytes"

/**
 * A screen in one state: what to compose, when it has left its loading state, and anything to
 * do before each frame (Settings re-requests its sub-page, which the screen consumes on show).
 */
internal class AuditScene(
    val ready: () -> Boolean = { true },
    val prepare: () -> Unit = {},
    val content: @Composable () -> Unit,
)

/**
 * Draws every [states] entry at every font scale in [fontScales] inside a [size] box, writing
 * `<state>-<w>x<h>-font<s>.png` under `AUDIT_OUT_ROOT/<screen>/`.
 *
 * Content goes through the activity's own `setContent`, as FirstLaunchRenderTest does: the
 * rule's `setContent` may be called once per test, and this loops. Each frame is `key`ed so a
 * screen starts fresh (scroll position, saved page). A state that never leaves its loading
 * state within [AUDIT_WAIT_MS] is still drawn — the picture then shows what it did instead —
 * and its file is listed in `AUDIT_OUT_ROOT/never-ready.txt`; it never fails the method. The
 * only assertion is [captureAuditFrame]'s: the file exists and is non-empty.
 */
internal fun AndroidComposeTestRule<*, out ComponentActivity>.renderAuditMatrix(
    screen: String,
    size: AuditSize,
    states: List<Pair<String, () -> AuditScene>>,
    fontScales: List<Float> = AUDIT_FONT_SCALES,
) {
    states.forEach { (state, build) ->
        val scene = build()
        fontScales.forEach { scale ->
            val name = auditFrameName(state, size, scale)
            scene.prepare()
            activity.setContent {
                val base = LocalDensity.current
                CompositionLocalProvider(
                    LocalWeightUnit provides WeightUnit.LBS,
                    LocalClockFormat provides ClockFormat.TWELVE,
                    LocalDensity provides Density(density = base.density, fontScale = scale),
                ) {
                    PersonalTrainerTheme(reduceMotion = false) {
                        key(name) {
                            Box(modifier = Modifier.width(size.w.dp).height(size.h.dp).background(Pit)) {
                                scene.content()
                            }
                        }
                    }
                }
            }
            val loaded = awaitAuditReady(scene.ready)
            waitForIdle()
            captureAuditFrame(screen, state, size, scale)
            if (!loaded) auditRootFile(AUDIT_NEVER_READY).appendText("$screen/$name.png\n")
        }
    }
}

/** `<state>-<w>x<h>-font<s>`: the frame's file name without its extension. */
internal fun auditFrameName(state: String, size: AuditSize, scale: Float): String =
    "$state-${size.w}x${size.h}-font$scale"

/** A file directly under [AUDIT_OUT_ROOT], the directory made if missing. */
internal fun auditRootFile(name: String): File = File(File(AUDIT_OUT_ROOT).apply { mkdirs() }, name)

/**
 * Polls [ready] for up to [AUDIT_WAIT_MS], idling the main looper on every poll; whether it
 * came true.
 *
 * `waitUntil` only sleeps between polls, and under Robolectric the test thread is the main
 * thread: a state a ViewModel posts to the main looper from Room's threads is never delivered
 * while the test sleeps, which is why run 1 saw the first frame of Plan day, Progress and
 * Summary stay in its loading state for the whole 20 s (the rule's own `setContent` hides this
 * by calling `waitForIdle()` first; the matrix composes through `activity.setContent`). So each
 * poll drains the looper as ExerciseDetailThisWorkoutRenderTest.tearDown and
 * RestTimerRecoveryTest do. The deadline is `System.nanoTime()`, as those tests use, not
 * `SystemClock`: Robolectric's clock only moves when the looper is idled by a duration, so a
 * deadline on it could never pass.
 */
internal fun AndroidComposeTestRule<*, out ComponentActivity>.awaitAuditReady(ready: () -> Boolean): Boolean {
    val giveUpAt = System.nanoTime() + AUDIT_WAIT_MS * 1_000_000L
    while (!ready() && System.nanoTime() < giveUpAt) {
        waitForIdle()
        Shadows.shadowOf(Looper.getMainLooper()).idle()
        Thread.sleep(AUDIT_POLL_MS)
    }
    return ready()
}

/**
 * Robolectric never delivers the draw callback `captureToImage` waits on; draw the window
 * ourselves and crop to the frame, as FirstLaunchRenderTest does. Every frame written is one
 * row of `AUDIT_OUT_ROOT/manifest.csv` (`file,screen,state,width,height,font,bytes`).
 */
internal fun AndroidComposeTestRule<*, out ComponentActivity>.captureAuditFrame(
    screen: String,
    state: String,
    size: AuditSize,
    scale: Float,
) {
    val name = auditFrameName(state, size, scale)
    val bitmap = runOnIdle {
        val decor = activity.window.decorView
        val full = Bitmap.createBitmap(
            decor.width.coerceAtLeast(1),
            decor.height.coerceAtLeast(1),
            Bitmap.Config.ARGB_8888,
        )
        decor.draw(Canvas(full))
        val px = decor.resources.displayMetrics.density
        Bitmap.createBitmap(
            full,
            0,
            0,
            (size.w * px).toInt().coerceIn(1, full.width),
            (size.h * px).toInt().coerceIn(1, full.height),
        )
    }
    val dir = File(AUDIT_OUT_ROOT, screen).apply { mkdirs() }
    val file = File(dir, "$name.png")
    file.outputStream().use { bitmap.compress(Bitmap.CompressFormat.PNG, 100, it) }
    assertTrue("$file must exist and be non-empty", file.exists() && file.length() > 0L)
    val manifest = auditRootFile(AUDIT_MANIFEST)
    if (!manifest.exists()) manifest.writeText("$AUDIT_MANIFEST_HEADER\n")
    manifest.appendText("$screen/$name.png,$screen,$state,${size.w},${size.h},$scale,${file.length()}\n")
}

/** Composes nothing, so the previous scene's ViewModels can be cancelled and their graph closed. */
internal fun AndroidComposeTestRule<*, out ComponentActivity>.blankAuditFrame() {
    activity.setContent { }
    waitForIdle()
}

@RunWith(RobolectricTestRunner::class)

    @get:Rule val compose = createAndroidComposeRule<ComponentActivity>()

    private var deps: FakeAppDependencies? = null
    private val viewModels = mutableListOf<ViewModel>()

    private val app: Application get() = ApplicationProvider.getApplicationContext()

    /** Cancelled on the main looper and drained there, as ExerciseDetailThisWorkoutRenderTest does. */
    @After
    fun tearDown() {
        closeGraph()
    }

    private fun closeGraph() {
        viewModels.forEach { it.viewModelScope.cancel() }
        Shadows.shadowOf(Looper.getMainLooper()).idle()
        viewModels.clear()
        deps?.close()
        deps = null
    }

    /** A fresh in-memory graph in pounds; the previous one, if any, is composed away and closed first. */
    private fun graph(
        insights: Flow<TrainingInsights> = MutableStateFlow(TrainingInsights()),
        activityDaoDecorator: (ActivityDao) -> ActivityDao = { it },
        workoutDaoDecorator: (WorkoutDao) -> WorkoutDao = { it },
    ): FakeAppDependencies {
        if (deps != null) compose.blankAuditFrame()
        closeGraph()
        return FakeAppDependencies(
            context = app,
            insights = insights,
            activityDaoDecorator = activityDaoDecorator,
            workoutDaoDecorator = workoutDaoDecorator,
        ).also {
            deps = it
            runBlocking { it.preferencesRepository.setWeightUnit(WeightUnit.LBS) }
        }
    }

    private fun <T : ViewModel> T.tracked(): T = also(viewModels::add)

    private fun matrix(screen: String, size: AuditSize, states: List<Pair<String, () -> AuditScene>>) =
        compose.renderAuditMatrix(screen, size, states)

    private val liveSessionId: String by lazy { runBlocking { seedTestWorkout(deps).session.id } }

    /** The rest page idle (no rest running) and on a running 90 s rest (RestPageSkipTest.openThePageOnARunningRest). */
    private fun restStates(): List<Pair<String, () -> AuditScene>> = listOf(
        "empty" to {
            deps.restTimerController.stop()
            val vm = restViewModel(liveSessionId)
            AuditScene(ready = { vm.uiState.value.loadState == SessionLoadState.FOUND }) {
                RestTimerScreen(onClose = {}, viewModel = vm)
            }
        },
        "running" to {
            val vm = restViewModel(liveSessionId)
            deps.restTimerController.start(90, liveSessionId)
            AuditScene(ready = { vm.uiState.value.loadState == SessionLoadState.FOUND && vm.uiState.value.rest.running }) {
                RestTimerScreen(onClose = {}, viewModel = vm)
            }
        },
    )

    private fun restViewModel(sessionId: String) = RestTimerViewModel(
        application = app,
        savedStateHandle = SavedStateHandle(mapOf("sessionId" to sessionId)),
        container = deps,
    ).also(restViewModels::add)
}

// ---------------------------------------------------------------------------------------------
// The shell. PersonalTrainerNav casts the application (AppNav.kt:293) and every screen inside it
// resolves its ViewModel through Application.appContainer(), so this runs under the real
// PersonalTrainerApp (RestLockSkipTest precedent) over the real AppContainer: an on-disk Room in
// Robolectric's per-test temp dir and the production DataStore delegate. First-run is the fresh
// app (Home behind its get-started sheet); Home is the same container with setup marked complete.
// ---------------------------------------------------------------------------------------------

@RunWith(RobolectricTestRunner::class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(application = PersonalTrainerApp::class, qualifiers = "w360dp-h640dp-mdpi")
class AuditShellRenderTest {
    @get:Rule val compose = createAndroidComposeRule<ComponentActivity>()

    private val app: PersonalTrainerApp get() = ApplicationProvider.getApplicationContext()

    @Test fun shell_360x640() = shell(AUDIT_360)

    @Test @Config(qualifiers = "w412dp-h915dp-mdpi")
    fun shell_412x915() = shell(AUDIT_412)

    @Test @Config(qualifiers = "w800dp-h360dp-land-mdpi")
    fun shell_800x360() = shell(AUDIT_800)

    @Test @Config(qualifiers = "w600dp-h960dp-mdpi")
    fun shell_600x960() = shell(AUDIT_600)

    private fun shell(size: AuditSize) {
        compose.renderAuditMatrix(
            "shell",
            size,
            listOf(
                "first-run" to { AuditScene(ready = ::homeTabShown) { PersonalTrainerNav() } },
                "home" to {
                    runBlocking { app.container.preferencesRepository.setOnboardingComplete(true) }
                    AuditScene(ready = ::homeTabShown) { PersonalTrainerNav() }
                },
            ),
        )
    }

    /** The gate has spoken and the bottom bar is up: its Home tab is on screen. */
    private fun homeTabShown(): Boolean = compose
        .onAllNodes(
            hasText("Home", substring = true) or hasContentDescription("Home", substring = true),
            useUnmergedTree = true,
        )
        .fetchSemanticsNodes()
        .isNotEmpty()
}
